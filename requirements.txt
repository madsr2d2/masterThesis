# Pinned to the versions the published numbers in every ANALYSIS.md were
# computed under, on Python 3.12. Loosen deliberately, not by accident: the
# archive-wide statistics are sensitive to floating-point detail in the

# the gate that would notice).

numpy==2.5.2

scipy==1.18.1
